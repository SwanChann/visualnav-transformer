\begin{nomenclature}
\label{chap:symb}

\begin{longtable}{rl}
  NoMaD           & Goal Masked Diffusion Policies for Navigation and Exploration \\
  DDPM            & Denoising Diffusion Probabilistic Model \\
  DDIM            & Denoising Diffusion Implicit Model \\
  CFG             & Classifier-Free Guidance \\
  TTS             & Test-Time Scaling \\
  GNM             & General Navigation Model \\
  ViNT            & Visual Navigation Transformer \\
  ViT             & Vision Transformer \\
  DiT             & Diffusion Transformer \\
  DINOv2          & Self-Distillation with No Labels v2 \\
  MuJoCo          & Multi-Joint dynamics with Contact \\
  SLAM            & Simultaneous Localization and Mapping \\
  PD              & Proportional-Derivative (Controller) \\
  topomap         & Topological Map（拓扑地图） \\
  waypoint        & 局部路径点 \\
  $\mathcal{O}_t$ & 时刻 $t$ 的历史视觉观察序列 \\
  $I_g$           & 目标图像或拓扑节点图像 \\
  $\mathbf{A}_t$  & 未来局部 waypoint 轨迹 \\
  $\mathbf{a}_i$  & 第 $i$ 个 waypoint，$(\Delta x_i, \Delta y_i)$ \\
  $H$             & 轨迹预测 horizon（waypoint 数量） \\
  $\beta_t$       & 扩散前向过程第 $t$ 步噪声调度系数 \\
  $\bar\alpha_t$  & 累计噪声调度系数 \\
  $\epsilon$      & 真实高斯噪声 \\
  $\epsilon_\theta$ & 模型预测噪声 \\
  $w$             & CFG guidance scale \\
  $N$             & TTS 候选轨迹数量 \\
  $\hat d_j$      & 距离预测头对第 $j$ 个候选节点的距离估计 \\
  cmd\_vel        & 速度控制指令 \\
  UDP             & User Datagram Protocol \\
  Orin            & NVIDIA Jetson Orin 边缘计算平台 \\
  Lite3           & 云深处 Lite3 四足机器人平台 \\
\end{longtable}

\end{nomenclature}
